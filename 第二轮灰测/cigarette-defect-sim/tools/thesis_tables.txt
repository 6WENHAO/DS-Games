算法 4-B ParallelConsumer(Q1 , Q2 , Q3 , Q4)
1:  save_pool ← ThreadPool(workers = WS_SAVE_WORKERS)
2:  last_activity ← Now();  is_idle ← False
3:  while not Stopped do
4:      WaitIfPaused();  items ← ∅
5:      // 公平轮询：每队列各取 ⌊B/4⌋ 张
6:      for each (cid, Q) in queues do
7:          for k = 1 to B/4 do try: items.Append((cid, Q.GetNoWait()))
8:      end for
9:      if items = g then
10:         if Now() 一 last_activity ≥ IDLE_THRESHOLD then
11:             is_idle ← True; PerfStats.Stop(); emit IdleChanged(True)
12:         end if
13:         Sleep(POLL_INTERVAL); continue
14:     end if
15:     if is_idle then is_idle ← False; PerfStats.Resume(); emit IdleChanged(False)
16:     last_activity ← Now();  imgs ← ParallelRead(items.paths)
17:     timer.Start(); results ← ModelManager.Predict(imgs, conf, iou, IMGSZ)
18:     timer.Stop(); per_ms ← timer.Elapsed() / |imgs|
19:     for each (cid, path, img, res) in zip(items, results) do
20:         (dets, sum, boxes, cls, confs) ← ExtractDetections(res)
21:         rendered ← DrawDetections(img, boxes, cls, confs, res.names)
22:         emit FrameReady(cid, rendered, info, per_ms)
23:         save_pool.Submit(SafeImWrite, cam_dir[cid]/path, rendered)
24:         emit ImageSaved(cid, save_path)
25:     end for
26: end while
